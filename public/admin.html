// Ambil data sesi admin saat halaman dibuka const adminSession = JSON.parse(sessionStorage.getItem("admin_session") || "{}"); $(document).ready(function() { // 1. Validasi Keamanan: Jika belum login sebagai admin, tendang ke index.html if (!adminSession.username || !adminSession.role) { alert("Akses ditolak! Silakan login sebagai Admin terlebih dahulu."); window.location.href = "/index.html"; return; } // 2. Tampilkan Nama & Peran Admin di Header Panel let infoRole = adminSession.role === 'PUSAT' ? 'Admin Pusat (Super Admin)' : `Admin Sekolah: ${adminSession.sekolah}`; $('#adminTitle').text(`Panel Management CBT - ${infoRole}`); // 3. Atur Fitur Berdasarkan Role (Role-Based Access Control) if (adminSession.role === 'SEKOLAH') { // Sembunyikan menu/tombol tertentu jika login sebagai Admin Sekolah $('#menu-upload-soal').hide(); // Sembunyikan upload soal jika soal terpusat (TKA) $('#fSekolah').val(adminSession.sekolah).prop('disabled', true); // Kunci filter ke sekolah sendiri } // Muat data dashboard pertama kali loadDashboard(); }); // Fungsi untuk mengirim role & sekolah saat mengambil data ke backend function getAdminParams() { return `role=${adminSession.role}&sekolah=${encodeURIComponent(adminSession.sekolah || '')}`; } function loadDashboard() { fetch(`/api/ujian?action=getDashboardData&${getAdminParams()}`) .then(r => r.json()) .then(res => { $('#stat-siswa').text(res.totalSiswa || 0); $('#stat-online').text(res.siswaOnline || 0); $('#stat-ujian').text(res.ujianAktif || 0); $('#stat-soal').text(res.totalSoal || 0); }); }
